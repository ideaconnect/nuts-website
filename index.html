Quick Example

Caddyfile

:8080 {
  route /events* {
    nuts {
      nats_url  nats://localhost:4222
      stream_name EVENTS
      topic_prefix events.
    }
  }
}

Client (JavaScript)

const es = new EventSource(
  '/events?topic=my-topic'
);

es.addEventListener('message', (e) => {
  const data = JSON.parse(e.data);
  console.log(data.topic, data.payload);
});
